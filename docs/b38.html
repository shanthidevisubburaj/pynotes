from sklearn.feature_extraction.text import CountVectorizer
from sklearn.naive_bayes import MultinomialNB


messages=[
"Win money now",
"Meeting at 5 pm",
"Free prize",
"Project discussion"
]


labels=[
"spam",
"normal",
"spam",
"normal"
]


vector=CountVectorizer()

x=vector.fit_transform(messages)


model=MultinomialNB()

model.fit(x,labels)

result=model.predict(
vector.transform([test])
)
